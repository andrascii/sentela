# nginx reverse proxy для Sentela: TLS-терминация + проксирование на app:3000.
# Сертификат/ключ монтируются с хоста (см. docker-compose.nginx.yml).

map $http_upgrade $connection_upgrade {
    default upgrade;
    ''      close;
}

# HTTPS
server {
    listen 127.0.0.1:8000 ssl;
    #listen [::]:8000 ssl;
    http2 on;
    server_name sentela.org www.sentela.org;

    ssl_certificate     /etc/monkeyisland/ssl/sentela.org/fullchain.pem;
    ssl_certificate_key /etc/monkeyisland/ssl/sentela.org/privatekey.pem;

    ssl_protocols       TLSv1.2 TLSv1.3;
    ssl_session_cache   shared:SSL:10m;
    ssl_session_timeout 1d;

    # Размер тела (конфиг API-мониторов и т.п.)
    client_max_body_size 2m;

    # Realtime WebSocket service (live dashboard updates) — separate Node process
    # published on host port 3001. Must come before "location /".
    location /realtime {
        proxy_pass http://127.0.0.1:3001;
        proxy_http_version 1.1;
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;
        proxy_set_header Upgrade           $http_upgrade;
        proxy_set_header Connection        $connection_upgrade;
        proxy_read_timeout 3600s;
    }

    location /monitor {
        proxy_pass http://127.0.0.1:10086;

        proxy_http_version 1.1;

        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection $connection_upgrade;
        proxy_set_header Host $host;

        proxy_read_timeout 86400;
        proxy_send_timeout 86400;

        proxy_buffering off;
        proxy_request_buffering off;
        proxy_ignore_client_abort on;

        tcp_nodelay on;
    }

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;
        proxy_set_header Upgrade           $http_upgrade;
        proxy_set_header Connection        $connection_upgrade;
        proxy_read_timeout 60s;
    }
}

server {
    listen 127.0.0.1:8080 ssl http2 proxy_protocol;

    server_name .monkeyisland.xyz;

    ssl_certificate /etc/monkeyisland/ssl/monkeyisland.xyz/fullchain.pem;
    ssl_certificate_key /etc/monkeyisland/ssl/monkeyisland.xyz/privatekey.pem;

    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_prefer_server_ciphers on;
    ssl_ciphers "ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384";

    ssl_stapling on;
    ssl_stapling_verify on;

    resolver 8.8.8.8 8.8.4.4 valid=300s;
    resolver_timeout 5s;

    # Настройки Proxy Protocol
    real_ip_header proxy_protocol;
    set_real_ip_from 127.0.0.1;

    location / {
        root /var/www/html;
        index index.html;
    }
}
